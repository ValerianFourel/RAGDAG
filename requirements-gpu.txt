# A100 / CUDA environment. REQUIRES PYTHON >= 3.10 (3.11 recommended).
#
# If `pip install -r` on this file reports "Could not find a version that
# satisfies scikit-learn" or similar, check `python -V` FIRST. On Python 3.9 pip
# aborts the whole file at the first unsatisfiable pin and installs nothing,
# which then looks like a dozen unrelated ModuleNotFoundErrors.
#
# Install order (login node, with internet):
#
#   curl -LsSf https://astral.sh/uv/install.sh | sh     # standalone CPython,
#   export PATH="$HOME/.local/bin:$PATH"                # no module system needed
#   uv venv --python 3.11 .venv && source .venv/bin/activate
#   uv pip install torch --index-url https://download.pytorch.org/whl/cu124
#   uv pip install -r requirements-gpu.txt
#
# The macOS x86_64 pins in requirements.txt (torch==2.2.2 and the versions it
# drags along) do NOT apply here — that ceiling exists only because PyTorch
# stopped shipping macOS Intel wheels after 2.2.2. On Linux+CUDA use current
# versions; torch 2.6.0+cu124 is known good.

torch>=2.4                    # any CUDA build; install from the cu12x index first
transformers>=4.44,<5         # 5.x needs torch>=2.5; 4.x keeps parity with the CPU run
sentence-transformers>=3.0,<4 # v5 changed the CrossEncoder API this code uses
tokenizers<0.21
huggingface-hub<1.0

scikit-learn>=1.4,<1.8
